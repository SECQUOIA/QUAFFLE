"""Local CPU construction/inference check for future full-size configurations."""
import copy
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess
import sys

ROOT = Path.cwd().resolve()
sys.path.insert(0, str(ROOT))
import torch
import centralized_demoTorchQ as quantum_entry
import ablation_studies as classical_entry
from scripts.run_local_spatial_study import study_specs
from scripts.collect_local_spatial_study import shared_parameter_digest
from tools import training_common as tc
from utilsTorch.utilsTorch import QVUNetSegmentation, ClassicalAblationUNetSegmentation

torch.set_num_threads(2)
torch.set_num_interop_threads(2)
common = ['--dataset', 'combined', '--seed', '42', '--base-channels', '32',
    '--image-size', '128', '--batch-size', '8', '--quantum_channels', '256',
    '--q-ansatz', 'reupload', '--q-layers', '1', '--n-qubits', '4',
    '--q-backend', 'pennylane', '--q-sim-device', 'cpu', '--local-spatial-channels', '4',
    '--decoder-skip-gate', 'none', '--skip-modulation', 'affine', '--no-time-emb',
    '--loss', 'dice_ce', '--epochs', '120', '--learning-rate', '0.001',
    '--eval-every', '2', '--no-save-samples', '--data-root', 'DATA_ROOT',
    '--results-dir', 'results_spatial_confirmation']
records, expected_shared = [], None
for spec in study_specs():
    options = spec['config']
    flags = [*common, '--local-spatial-kernel', options['local_spatial_kernel'],
        '--local-spatial-readout', options['local_spatial_readout'],
        '--q-training-control', options.get('q_training_control', 'trainable')]
    if spec['model_type'] == 'quantum':
        script = 'centralized_demoTorchQ.py'
        sys.argv = [script, *flags]
        args = quantum_entry.parse_args()
        config = dict(quantum_entry.get_config(), seed=args.seed, dataset=args.dataset)
        tc.apply_common_args(config, args)
        cls = QVUNetSegmentation
    else:
        script = 'ablation_studies.py'
        flags = ['--framework', 'torch', '--mode', 'centralized',
            '--quantum-backend', 'pennylane', '--ablation-mlp-mode', 'full_rank', *flags]
        sys.argv = [script, *flags]
        args = classical_entry.parse_args()
        config, counted_classical, counted_quantum = classical_entry._find_matched_torch_config(
            args.quantum_backend, classical_entry._torch_overrides(args))
        cls = ClassicalAblationUNetSegmentation
    tc.seed_everything(42)
    model = tc.build_model(cls, copy.deepcopy(config)).to('cpu').eval()
    shared, core = shared_parameter_digest(model)
    if expected_shared is not None:
        assert shared == expected_shared, 'Full-size controls have different shared initialization'
    expected_shared = shared
    images = torch.randn((1, 3, 128, 128), generator=torch.Generator().manual_seed(123))
    with torch.no_grad():
        output = model(images)
    assert tuple(output.shape) == (1, 2, 128, 128) and bool(torch.isfinite(output).all())
    record = dict(id=spec['id'], production_command=['python', script, *flags],
        base_channels=config['base_channels'], quantum_channels=config['quantum_channels'],
        resnet_block_groups=config['resnet_block_groups'], image_size=config['image_size'],
        configured_batch_size=config['batch_size'], probe_batch_size=1,
        total_params=sum(p.numel() for p in model.parameters()),
        trainable_params=sum(p.numel() for p in model.parameters() if p.requires_grad),
        core_params=core, shared_initial_parameter_sha256=shared,
        output_shape=list(output.shape), finite_random_tensor_inference=True)
    if spec['model_type'] == 'ablation':
        assert record['total_params'] == counted_classical
        record['quantum_reference_params'] = counted_quantum
    records.append(record)
    print(json.dumps(record), flush=True)
    del model, output, images

plan = json.loads(Path('/tmp/quaffle-local-spatial-pilot60-20261027/plan.json').read_text())
for name, expected in plan['source_hashes'].items():
    assert hashlib.sha256((ROOT / name).read_bytes()).hexdigest() == expected
result = dict(source_revision=subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip(),
    producer_source_hashes=plan['source_hashes'],
    ablation_entrypoint_sha256=hashlib.sha256((ROOT / 'ablation_studies.py').read_bytes()).hexdigest(),
    probe_source_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    completed_at=datetime.now(timezone.utc).isoformat(), device='cpu', threads=2,
    scope='Seven production CLI/configuration paths; full-size model initialization and one random batch-1 inference. No data, optimization, accuracy, Anvil execution, or GPU memory benchmark.',
    records=records)
target = Path('/tmp/quaffle-local-spatial-anvil-config-audit.json')
assert not target.exists()
target.write_text(json.dumps(result, indent=2) + '\n')
