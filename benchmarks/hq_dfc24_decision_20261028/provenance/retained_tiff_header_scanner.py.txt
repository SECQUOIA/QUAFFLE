"""Metadata-only scan of retained DFC24 TIFF copies; no raster decoding.

The sizes/geo_tags/text_names definitions and tiff_metadata function below are
the exact parser fragment executed for local_dfc24_exposure_audit.json. The CLI
is a retention wrapper; it does not inspect upstream datasets or infer events.
"""
import argparse
import hashlib
import json
from pathlib import Path
import struct


sizes={1:1,2:1,3:2,4:4,5:8,6:1,7:1,8:2,9:4,10:8,11:4,12:8,13:4,16:8,17:8,18:8}
geo_tags={33550,33922,34264,34735,34736,34737,34853,40965}
text_names={270:'ImageDescription',305:'Software',306:'DateTime',315:'Artist',316:'HostComputer',
 33432:'Copyright',42112:'GDALMetadata',42113:'GDALNoData',700:'XMP'}
def tiff_metadata(path):
 with path.open('rb') as stream:
  head=stream.read(16);endian='<' if head[:2]==b'II' else '>' if head[:2]==b'MM' else None
  assert endian,path
  version=struct.unpack(endian+'H',head[2:4])[0]
  if version==42:offset=struct.unpack(endian+'I',head[4:8])[0];count_fmt='H';entry_size=12;pointer_size=4;entry_fmt='HHI4s'
  elif version==43:
   assert struct.unpack(endian+'HH',head[4:8])==(8,0)
   offset=struct.unpack(endian+'Q',head[8:16])[0];count_fmt='Q';entry_size=20;pointer_size=8;entry_fmt='HHQ8s'
  else:raise ValueError('Unsupported TIFF version '+str(version))
  tags=set();texts={};geometry={};seen=set();ifds=0
  while offset:
   assert offset not in seen and len(seen)<2048
   seen.add(offset);ifds+=1;stream.seek(offset);count=struct.unpack(endian+count_fmt,stream.read(struct.calcsize(count_fmt)))[0]
   assert count<4096
   entries=[struct.unpack(endian+entry_fmt,stream.read(entry_size)) for _ in range(count)]
   next_offset=struct.unpack(endian+('I' if pointer_size==4 else 'Q'),stream.read(pointer_size))[0]
   for tag,kind,n,value in entries:
    tags.add(tag)
    if kind not in sizes:continue
    if tag not in (256,257,277) and kind!=2 and tag not in text_names and tag not in geo_tags:continue
    byte_count=n*sizes[kind]
    if byte_count<=pointer_size:raw=value[:byte_count]
    else:
     stream.seek(struct.unpack(endian+('I' if pointer_size==4 else 'Q'),value)[0]);raw=stream.read(min(byte_count,65536))
    if tag in (256,257,277) and n==1:
     fmt={3:'H',4:'I',16:'Q'}.get(kind)
     if fmt:geometry[str(tag)]=struct.unpack(endian+fmt,raw)[0]
    if kind==2 or tag in text_names:
     texts[str(tag)]=dict(name=text_names.get(tag,'ASCII'),declared_bytes=byte_count,
       text=raw.rstrip(b'\x00').decode('utf-8',errors='replace'),truncated=byte_count>len(raw))
   offset=next_offset
  return dict(version=version,ifd_count=ifds,tags=sorted(tags),geometry=geometry,text_metadata=texts,
              geolocation_tag_ids=sorted(tags & geo_tags))


def scan_retained_dataset(dataset_root):
    root = Path(dataset_root).resolve()
    headers = []
    for path in sorted((root/'flood_combined').rglob('*.tif')):
        info = tiff_metadata(path)
        info['path'] = str(path.relative_to(root))
        headers.append(info)
    if not headers:
        raise ValueError('No retained combined TIFFs found')
    canonical = json.dumps(headers, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()
    return dict(scope='Retained local TIFF copies only; no upstream georeferencing/event metadata inference.',
        dataset_root=str(root), files=len(headers), ifds=sum(item['ifd_count'] for item in headers),
        geolocation_tag_files=sum(bool(item['geolocation_tag_ids']) for item in headers),
        text_metadata_entries=sum(len(item['text_metadata']) for item in headers),
        headers_sha256=hashlib.sha256(canonical).hexdigest(), headers=headers)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--dataset-root', type=Path, required=True)
    parser.add_argument('--output', type=Path, required=True)
    args = parser.parse_args()
    if args.output.exists():
        parser.error('Refusing to overwrite retained scanner evidence')
    if not args.output.parent.is_dir():
        parser.error('Output parent must already exist')
    result = scan_retained_dataset(args.dataset_root)
    args.output.write_text(json.dumps(result, indent=2, sort_keys=True, allow_nan=False)+'\n')
    print(json.dumps({key: value for key, value in result.items() if key != 'headers'}, indent=2))


if __name__ == '__main__':
    main()
