class NotComplete(ValueError):
    pass


def need(value, message):
    if not value:
        raise ValueError(message)


def sha(path):
    result = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1024*1024), b''):
            result.update(block)
    return result.hexdigest()


def bounded_json(path, limit=4*1024*1024):
    path = Path(path)
    need(path.is_file() and 0 < path.stat().st_size <= limit, 'Missing/oversized JSON: '+str(path))
    def unique(pairs):
        value = {}
        for key, child in pairs:
            need(key not in value, 'Duplicate JSON key: '+key)
            value[key] = child
        return value
    def invalid(value):
        raise ValueError('Nonfinite JSON constant: '+value)
    value = json.loads(path.read_text(), object_pairs_hook=unique, parse_constant=invalid)
    budget = [300000]
    def walk(child, depth):
        budget[0] -= 1
        need(budget[0] >= 0 and depth <= 24, 'Oversized/nested JSON structure')
        if isinstance(child, float):
            need(math.isfinite(child), 'Nonfinite JSON number')
        elif isinstance(child, dict):
            need(len(child) <= 20000, 'Oversized JSON mapping')
            for item in child.values(): walk(item, depth+1)
        elif isinstance(child, list):
            need(len(child) <= 20000, 'Oversized JSON list')
            for item in child: walk(item, depth+1)
        elif isinstance(child, str):
            need(len(child) <= 16384, 'Oversized JSON string')
        else:
            need(child is None or type(child) in (bool, int), 'Unsupported JSON value')
    walk(value, 0)
    return value


def inside(root, relative):
    need(isinstance(relative, str) and not Path(relative).is_absolute(), 'Expected relative artifact path')
    path = (root/relative).resolve()
    need(path != root and path.is_relative_to(root), 'Artifact path escapes its root')
    return path


def close(actual, expected, label):
    need(type(actual) in (int, float) and math.isfinite(actual), 'Invalid number: '+label)
    need(math.isclose(actual, expected, rel_tol=1e-13, abs_tol=2e-12),
         f'Numeric mismatch {label}: {actual!r} != {expected!r}')


def compare(actual, expected, label):
    if type(expected) is float:
        close(actual, expected, label)
    elif isinstance(expected, dict):
        need(isinstance(actual, dict) and set(actual) == set(expected), 'Mapping fields differ: '+label)
        for key in expected: compare(actual[key], expected[key], label+'/'+key)
    elif isinstance(expected, list):
        need(isinstance(actual, list) and len(actual) == len(expected), 'List size differs: '+label)
        for i, value in enumerate(expected): compare(actual[i], value, f'{label}/{i}')
    else:
        need(type(actual) is type(expected) and actual == expected, 'Typed value differs: '+label)


def metrics(matrix):
    need(isinstance(matrix, list) and len(matrix) == 5 and
         all(isinstance(row, list) and len(row) == 5 for row in matrix), 'Expected a 5x5 confusion matrix')
    need(all(type(value) is int and 0 <= value < 2**50 for row in matrix for value in row),
         'Confusion counts must be bounded nonnegative integers')
    total = sum(sum(row) for row in matrix)
    need(total > 0, 'Empty confusion matrix')
    truth = [sum(row) for row in matrix]
    predicted = [sum(matrix[row][column] for row in range(5)) for column in range(5)]
    unions = [truth[i]+predicted[i]-matrix[i][i] for i in range(5)]
    ious = [Fraction(matrix[i][i], unions[i]) if unions[i] else Fraction(0) for i in range(5)]
    return dict(confusion=matrix, pixels=total, truth_pixels=truth, prediction_pixels=predicted,
        class_iou=dict(zip(CLASS_NAMES, map(float, ious))),
        absent_union=[CLASS_NAMES[i] for i in range(5) if unions[i] == 0],
        miou=float(sum(ious, Fraction())/5), foreground_miou=float(sum(ious[1:], Fraction())/4),
        accuracy=float(Fraction(sum(matrix[i][i] for i in range(5)), total)),
        exact_miou=sum(ious, Fraction())/5)


def check_observation(observed, label):
    result = metrics(observed['confusion'])
    need(result['truth_pixels'] == VAL_COUNTS and result['pixels'] == 64*512*512,
         'Validation truth/pixel cohort differs: '+label)
    for key, expected in result.items():
        if key != 'exact_miou': compare(observed[key], expected, label+'/'+key)
    need(type(observed['examples']) is int and observed['examples'] == 1024, 'Incomplete validation crops')
    need(type(observed['loss']) in (int, float) and observed['loss'] >= 0, 'Invalid validation loss')
    return result


def moments(values):
    need(len(values) >= 2, 'Sample SD needs two observations')
    mean = math.fsum(values)/len(values)
    variance = math.fsum((value-mean)**2 for value in values)/(len(values)-1)
    return mean, math.sqrt(variance)


def exact_screen(scores):
    contrasts = {reference: [scores['quantum', seed]-scores[reference, seed] for seed in SEEDS]
                 for reference in REFERENCES}
    threshold = {reference: sum(contrasts[reference], Fraction())/3 >= Fraction(1, 100)
                 for reference in REFERENCES[:2]}
    positive = {reference: all(delta > 0 for delta in contrasts[reference]) for reference in REFERENCES[:2]}
    zero_nonnegative = sum(contrasts['zero_bottleneck'], Fraction()) >= 0
    return dict(passed=all(threshold.values()) and all(positive.values()) and zero_nonnegative,
                at_least_one_pp=threshold, all_seeds_positive=positive,
                zero_mean_nonnegative=zero_nonnegative,
                exact_mean_differences={key: str(sum(values, Fraction())/3) for key, values in contrasts.items()})


def pick_rate(candidates):
    need(len(candidates) == 2 and [row['rate'] for row in candidates] == [.001, .003], 'Wrong affine tuning candidates')
    return min(candidates, key=lambda row: (-row['best_miou'], row['rate']))['rate']


def fit_name(row):
    return f"{row['control']}_seed{row['seed']}_lr{row['rate']:g}"


def identity(row):
    need(row['control'] in CONTROLS and type(row['seed']) is int and row['seed'] in SEEDS and
         type(row['rate']) is float and row['rate'] in (.001, .003), 'Wrong typed fit identity')
    return row['control'], row['seed'], row['rate']


def file_inventory(root, limit=100):
    files = []
    for path in root.rglob('*'):
        if path.is_file():
            files.append(path)
            need(len(files) <= limit, 'Unexpected artifact inventory size')
    files.sort()
    need(all(path.resolve().is_relative_to(root) for path in files), 'Artifact symlink escapes input root')
    return {str(path.relative_to(root)): sha(path) for path in files}


def check_csv(path, expected):
    need(path.stat().st_size <= 8*1024*1024 and expected, 'CSV size/expected rows invalid')
    with path.open(newline='') as stream:
        rows = list(csv.reader(stream))
    fields = list(expected[0])
    need(rows and rows[0] == fields and len(fields) == len(set(fields)), 'CSV header differs: '+path.name)
    need(len(rows) == len(expected)+1 and len(rows) <= 4097, 'CSV row count differs: '+path.name)
    for n, (row, want) in enumerate(zip(rows[1:], expected), 1):
        need(len(row) == len(fields), 'CSV header/row width differs: '+path.name)
        for field, cell in zip(fields, row):
            value = want[field]
            label = f'{path.name}/{n}/{field}'
            if type(value) is float:
                need(cell and not any(c.isspace() for c in cell), 'Invalid float CSV cell: '+label)
                close(float(cell), value, label)
            else:
                need(cell == ('' if value is None else str(value)), 'CSV value differs: '+label)
    return dict(columns=fields, rows=len(expected), all_cells_checked=True)


def complete_gate(task):
    # Deliberately the ONLY two reads before admission; no fits/exports/images.
    try:
        pipeline = bounded_json(task/'pipeline_status.json', 64*1024)
        status = bounded_json(task/'campaign_v1/status.json', 64*1024)
    except (OSError, ValueError) as error:
        raise NotComplete('Complete pipeline/campaign prerequisites absent: '+str(error)) from error
    if not (pipeline.get('stage') == 'complete' and pipeline.get('plan_sha') == PLAN_SHA and
            status.get('status') == 'complete' and type(status.get('fits')) is int and status.get('fits') == 13
            and status.get('plan_sha') == PLAN_SHA):
        raise NotComplete('NOT_COMPLETE: require pipeline stage complete and all 13 completed fits')
    return pipeline, status

def derived_tables(plan, fits, primary, chosen, observed):
    arms, pairs = [], []
    for control in CONTROLS:
        row = dict(control=control)
        for metric in ('miou', 'accuracy', 'foreground_miou'):
            row[metric+'_mean'], row[metric+'_sd'] = moments([observed[control, seed][metric] for seed in SEEDS])
        arms.append(row)
    for reference in REFERENCES:
        deltas = [100*(observed['quantum', seed]['miou']-observed[reference, seed]['miou']) for seed in SEEDS]
        mean, sd = moments(deltas)
        follow_mean, follow_sd = moments(deltas[1:])
        pairs.append(dict(reference=reference, metric='miou', seed42_pp=deltas[0], seed77_pp=deltas[1], seed123_pp=deltas[2],
            mean_pp=mean, sample_sd_pp=sd, positive_seeds=sum(value > 0 for value in deltas), calibration_seed42_pp=deltas[0],
            followup_seeds77_123_mean_pp=follow_mean, followup_seeds77_123_sample_sd_pp=follow_sd))
    exact = exact_screen({key: value['exact_miou'] for key, value in observed.items()})
    no_gain = sum(row['best_miou'] <= row['history'][0]['validation']['miou'] for row in primary)
    late = sum(row['best_epoch'] >= 55 for row in primary)
    summary = dict(arms=arms, paired=pairs, practical_screen_passed=exact['passed'],
        screening_scope='Descriptive fixed practical screen, not significance, equivalence or quantum advantage',
        initial_to_best_nonpositive_count=no_gain, all_arms_no_learning=no_gain == 12, late_best_count=late,
        horizons_unresolved=late > 0, training_quality='no_observed_learning' if no_gain == 12 else 'learning_observed')
    tables = {name: [] for name in CSV_NAMES}
    tables['paired'], tables['arm_summary'] = pairs, arms
    for row in primary:
        value = observed[row['control'], row['seed']]
        base = dict(control=row['control'], seed=row['seed'], rate=row['rate'])
        tables['per_seed'].append(dict(**base, best_epoch=row['best_epoch'], miou=value['miou'], accuracy=value['accuracy'],
            foreground_miou=value['foreground_miou'], examples=1024, pixels=16777216,
            checkpoint_sha=row['checkpoint_sha'], optimizer_steps=3840))
        for i, name in enumerate(CLASS_NAMES):
            tp = value['confusion'][i][i]
            tables['class_metrics'].append(dict(**base, class_id=i, class_name=name, iou=value['class_iou'][name],
                truth_pixels=value['truth_pixels'][i], prediction_pixels=value['prediction_pixels'][i], tp=tp,
                fp=value['prediction_pixels'][i]-tp, fn=value['truth_pixels'][i]-tp))
        for point in row['history']:
            score = metrics(point['validation']['confusion'])
            tables['curves'].append(dict(**base, epoch=point['epoch'], optimizer_steps=point['optimizer_steps'],
                train_loss=point['train_loss'], validation_loss=point['validation']['loss'], miou=score['miou'],
                accuracy=score['accuracy'], foreground_miou=score['foreground_miou'], selected=point['epoch'] == row['best_epoch']))
        for point in row['training_history']:
            tables['training_curves'].append(dict(**base, epoch=point['epoch'], optimizer_steps=point['optimizer_steps'], train_loss=point['train_loss']))
        initial = metrics(row['history'][0]['validation']['confusion'])['miou']
        best = metrics(next(point['validation']['confusion'] for point in row['history'] if point['epoch'] == row['best_epoch']))['miou']
        final = metrics(row['history'][-1]['validation']['confusion'])['miou']
        before_final = metrics(row['history'][-2]['validation']['confusion'])['miou']
        tables['learning'].append(dict(**base, initial_miou=initial, best_miou=best, final_miou=final,
            initial_to_best_pp=100*(best-initial), best_epoch=row['best_epoch'], best_at_cap=row['best_epoch'] == 60,
            best_at_or_after_55=row['best_epoch'] >= 55, last_interval_improvement_pp=100*(final-before_final)))
        tables['fit_costs'].append(dict(**base, elapsed_seconds=row['elapsed_seconds'],
            peak_cuda_allocated=row['peak_cuda_allocated'], **row['counts']))
    tables['rate_candidates'] = [dict(control=row['control'], seed=row['seed'], rate=row['rate'], best_miou=row['best_miou'],
        best_epoch=row['best_epoch'], selected_common_rate=row['rate'] == chosen) for row in fits
        if (row['control'], row['seed']) == ('compact_affine', 42)]
    for control, arm in plan['profile']['arms'].items():
        tables['profile_costs'].append(dict(control=control, warmup_seconds=arm['warmup_seconds'],
            step1_seconds=arm['step_seconds'][0], step2_seconds=arm['step_seconds'][1], eval_batch_seconds=arm['eval_batch_seconds'],
            peak_cuda_allocated=arm['peak_cuda_allocated'], peak_cuda_reserved=arm['peak_cuda_reserved'], **arm['counts']))
    return summary, tables, exact


def verify_export(root, plan, status, complete, replay, expected_summary, tables):
    actual = file_inventory(root)
    need(set(actual) == EXPORT_NAMES and len(actual) == 17, 'Expected exactly17 exported files')
    need(bounded_json(root/'artifact_sha256.json') == {name: value for name, value in actual.items() if name != 'artifact_sha256.json'},
         'Declared export SHA inventory differs')
    compare(bounded_json(root/'summary.json'), expected_summary, root.name+'/summary')
    for name, rows in tables.items(): check_csv(root/(name+'.csv'), rows)
    provenance = bounded_json(root/'provenance.json')
    need(provenance['format'] == 'landcover-native-export-v1' and provenance['plan_sha'] == PLAN_SHA and
         provenance['collector_sha'] == plan['source']['scripts/collect_landcover_native.py'] and
         provenance['all_fits_sha'] == status['all_fits_sha'] and provenance['replay_complete_sha'] == sha(replay/'complete.json') and
         provenance['replay_inventory_sha'] == sha(replay/'artifact_sha256.json') and provenance['source'] == plan['source'] and
         provenance['inference_performed'] is False and provenance['test_predictions'] == 0, 'Export provenance binding differs')
    need(sha(root/'collector_source.py.txt') == provenance['collector_sha'], 'Inert collector source differs')
    return actual


def selfcheck():
    toy = [[5, 1, 0, 0, 0], [2, 4, 0, 0, 0], [0, 0, 3, 1, 0], [0, 0, 1, 2, 1], [0, 0, 0, 0, 5]]
    value = metrics(toy)
    need(value['truth_pixels'] == [6, 6, 4, 4, 5] and value['prediction_pixels'] == [7, 5, 4, 3, 6], 'Orientation selfcheck')
    need(value['exact_miou'] == Fraction(509, 840) and value['accuracy'] == 19/25 and
         value['foreground_miou'] == float(Fraction(101, 168)), 'Hand-computed metric selfcheck')
    need(list(value['class_iou'].values()) == [5/8, 4/7, 3/5, 2/5, 5/6], 'Hand-computed class IoUs')
    empty_classes = [[int(i == j == 0) for j in range(5)] for i in range(5)]
    need(metrics(empty_classes)['exact_miou'] == Fraction(1, 5), 'Zero-union class convention')
    need(moments([1., 2., 3.]) == (2., 1.), 'Sample SD n-1 selfcheck')
    close(moments([2., 3.])[1], math.sqrt(.5), 'Two-followup sample SD')
    def cycle(diagonal, off_diagonal):
        return [[diagonal if i == j else off_diagonal if j == (i+1)%5 else 0 for j in range(5)] for i in range(5)]
    q, reference = cycle(58, 71), cycle(56, 72)
    need(metrics(q)['exact_miou'] == Fraction(29, 100) and metrics(reference)['exact_miou'] == Fraction(28, 100), 'Exact1pp matrices')
    scores = {(control, seed): metrics(q if control in ('quantum', 'zero_bottleneck') else reference)['exact_miou']
              for control in CONTROLS for seed in SEEDS}
    screen = exact_screen(scores)
    need(screen['passed'] and screen['zero_mean_nonnegative'], 'Exact1pp plus zero-mean equality must pass')
    one_pixel = [row[:] for row in q]
    one_pixel[0][0] -= 1; one_pixel[0][1] += 1
    lower = metrics(one_pixel)['exact_miou']
    below = {key: lower if key[0] in ('quantum', 'zero_bottleneck') else value for key, value in scores.items()}
    need(not exact_screen(below)['passed'] and all(exact_screen(below)['all_seeds_positive'].values())
         and exact_screen(below)['zero_mean_nonnegative'], 'One-pixel just-below boundary must fail threshold only')
    # All-positive is independently necessary even when the paired mean exceeds1pp.
    tie = dict(scores)
    for seed in (77, 123): tie['compact_affine', seed] = Fraction(27, 100)
    tie['compact_affine', 42] = tie['quantum', 42]
    need(exact_screen(tie)['at_least_one_pp']['compact_affine'] and not exact_screen(tie)['passed'], 'One tied seed must fail')
    need(pick_rate([dict(rate=.001, best_miou=.5), dict(rate=.003, best_miou=.5)]) == .001, 'Lower-LR exact tie')
    invalid_cases = [True, -1, 2**50]
    for invalid in invalid_cases:
        corrupt = [row[:] for row in toy]; corrupt[0][0] = invalid
        try: metrics(corrupt)
        except ValueError: pass
        else: raise AssertionError('Typed/bounded count refusal missing')
    from unittest.mock import patch
    for stage, completed_fits in [('waiting_for_live_training_unit', 13), ('complete', 12)]:
        accesses = []
        def fake_read(path, limit):
            accesses.append(str(path))
            if path.name == 'pipeline_status.json': return dict(stage=stage, plan_sha=PLAN_SHA)
            if path.name == 'status.json': return dict(status='complete', fits=completed_fits, plan_sha=PLAN_SHA)
            raise AssertionError('Admission opened a fit/plan/export before completion')
        with patch.object(sys.modules[__name__], 'bounded_json', side_effect=fake_read):
            try: audit(Path('/synthetic_numeric_audit_task'), Path('/synthetic_source'))
            except NotComplete: pass
            else: raise AssertionError('Incomplete pipeline admission must fail')
        need(accesses == ['/synthetic_numeric_audit_task/pipeline_status.json',
                          '/synthetic_numeric_audit_task/campaign_v1/status.json'], 'Read-before-admission guard failed')
    return dict(status='passed', hand_confusion=toy, exact_miou='509/840', accuracy='19/25',
        secondary_foreground_miou='101/168', sample_sd_n_minus_1=True, exact_one_pp_pass=True,
        one_pixel_below_fails=True, exact_zero_mean_equality_pass=True, positive_seed_tie_fails=True,
        lower_lr_tie_selects_001=True, invalid_boolean_negative_oversized_counts_rejected=True,
        incomplete_pipeline_and_fit_grid_stop_before_any_plan_fit_or_export_read=True,
        actual_campaign_outputs_read=False, project_helpers_imported=False)

def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--selfcheck', action='store_true')
    parser.add_argument('--task-root', type=Path, default=Path(__file__).resolve().parents[1])
    parser.add_argument('--source-root', type=Path, default=Path('/tmp/quaffle-landcover-native-20261028'))
    parser.add_argument('--output-dir', type=Path)
    args = parser.parse_args(argv)
    if args.selfcheck:
        print(json.dumps(selfcheck(), sort_keys=True, indent=2)); return 0
    if args.output_dir is None: parser.error('--output-dir is required for an authorized completed-pipeline audit')
    task, output = args.task_root.resolve(), args.output_dir.resolve()
    need(output.is_relative_to(task/'numeric_audit') and output != task/'numeric_audit', 'Output must stay under task numeric_audit/')
    need(not output.exists() or (output.is_dir() and not any(output.iterdir())), 'Preserve existing audit output')
    try:
        result = audit(task, args.source_root.resolve())
    except NotComplete as error:
        print(json.dumps(dict(status='NOT_COMPLETE', reason=str(error), actual_audit_started=False)), file=sys.stderr)
        return 3
    result['audit_source_sha256'] = sha(__file__)
    output.mkdir(parents=True, exist_ok=True)
    (output/'audit.json').write_text(json.dumps(result, sort_keys=True, indent=2, allow_nan=False)+'\n')
    (output/'REPORT.md').write_text(report(result))
    print(json.dumps(dict(status='passed', audit_sha256=sha(output/'audit.json'), report_sha256=sha(output/'REPORT.md'))))
    return 0
