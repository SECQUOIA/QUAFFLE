#!/usr/bin/env python3
"""Stage an allowlisted, byte-preserving publication payload after completion.

Stdlib only. This helper never imports scientific code, reads weights/images,
launches a process, computes scientific outcomes, or publishes. Admission checks
precede all result/history/export reads. Existing output paths are preserved.
"""
import argparse
import hashlib
import json
import math
from pathlib import Path

TASK = Path('/home/bernalde/repos/QUAFFLE/results/landcover_native_20261028')
PLAN_SHA = 'a3cae54c1f575fc46722513c8d274daf96ef4b3fc4635c4a3e051c7f6e1b30fc'
CSV_NAMES = ('per_seed','class_metrics','paired','arm_summary','curves','training_curves',
             'fit_costs','learning','rate_candidates','profile_costs')
EXPORT_NAMES = {name+'.csv' for name in CSV_NAMES} | {
    'summary.json','learning_curves.pdf','learning_curves.png','INTERPRETATION.md',
    'collector_source.py.txt','provenance.json','artifact_sha256.json'}
CONTROLS = ('compact_affine','quantum','frozen_quantum','zero_bottleneck')
SEEDS = (42,77,123)

def need(value,message):
    if not value:raise ValueError(message)


def sha(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda:stream.read(1024*1024),b''):digest.update(block)
    return digest.hexdigest()


def file_path(root,name):
    name = Path(name)
    need(not name.is_absolute() and '..' not in name.parts,'Unsafe relative member path')
    candidate = root/name
    need(candidate.is_file() and not candidate.is_symlink() and candidate.resolve().is_relative_to(root),
         'Missing, linked or escaping input: '+str(candidate))
    return candidate


def read(path,limit=8*1024*1024):
    path = Path(path)
    need(path.is_file() and not path.is_symlink() and 0 < path.stat().st_size <= limit,
         'Missing, linked or oversized JSON: '+str(path))
    def unique(items):
        result = {}
        for key,value in items:
            need(key not in result,'Duplicate JSON key: '+key);result[key]=value
        return result
    def invalid(value):raise ValueError('Nonfinite JSON: '+value)
    result = json.loads(path.read_text(),object_pairs_hook=unique,parse_constant=invalid)
    def finite(value,depth=0):
        need(depth <= 32,'Excessively nested JSON')
        if isinstance(value,float):need(math.isfinite(value),'Nonfinite JSON number')
        elif isinstance(value,dict):
            for item in value.values():finite(item,depth+1)
        elif isinstance(value,list):
            for item in value:finite(item,depth+1)
    finite(result)
    return result


def inventory(root,max_files=150):
    paths = sorted(path for path in root.rglob('*') if path.is_file() or path.is_symlink())
    need(len(paths) <= max_files,'Unexpected input inventory size')
    return {str(path.relative_to(root)):sha(file_path(root,path.relative_to(root))) for path in paths}


def gate(task):
    # Do not open plans, partial outcomes, histories or exports before BOTH
    # the completed production pipeline and the independent audit admit access.
    pipeline = read(task/'pipeline_status.json',64*1024)
    need(pipeline.get('stage') == 'complete' and pipeline.get('plan_sha') == PLAN_SHA,
         'NOT_READY: production pipeline is not complete for the frozen plan')
    campaign = read(task/'campaign_v1/status.json',64*1024)
    need(campaign.get('status') == 'complete' and campaign.get('fits') == 13 and
         campaign.get('plan_sha') == PLAN_SHA and campaign.get('test_predictions') == 0,
         'NOT_READY: require all thirteen completed fits with no test predictions')
    watcher = read(task/'numeric_audit/watcher_status.json',64*1024)
    need(watcher.get('stage') == 'complete','NOT_READY: independent numeric audit is not complete')
    audit_dir = task/'numeric_audit/completed_v1'
    audit = read(audit_dir/'audit.json')
    need(audit.get('status') == 'passed' and audit.get('plan_sha256') == PLAN_SHA and
         audit.get('raw_fit_count') == 13 and audit.get('primary_count') == 12 and audit.get('completed_replays') == 12,
         'NOT_READY: independent audit did not pass the complete thirteen-fit/twelve-replay study')
    need(sha(audit_dir/'audit.json') == watcher['audit_sha'] and sha(audit_dir/'REPORT.md') == watcher['report_sha'],
         'Independent audit/report byte bindings differ')
    oracle = task/'numeric_audit/independent_numeric_audit.py'
    need(sha(oracle) == audit['audit_source_sha256'] == watcher['oracle_sha'],'Independent audit source binding differs')
    return pipeline,campaign,watcher,audit


def fit_name(row):
    need(row['control'] in CONTROLS and type(row['seed']) is int and row['seed'] in SEEDS and
         type(row['rate']) in (int,float) and row['rate'] in (.001,.003),'Unexpected fit identity')
    return f"{row['control']}_seed{row['seed']}_lr{row['rate']:g}"


def make_allowlist(task):
    pipeline,status,watcher,audit = gate(task)
    campaign,replay = task/'campaign_v1',task/'replay_v1'
    plan = read(campaign/'plan.json');verification = read(task/'pipeline_verification.json')
    need(sha(campaign/'plan.json') == PLAN_SHA and sha(task/'pipeline_verification.json') == pipeline['verification_sha'],
         'Frozen plan/operational completion SHA differs')
    need(verification['plan_sha'] == PLAN_SHA and verification['source'] == plan['source'] == audit['source_hashes'] and
         verification['fit_count'] == 13 and verification['selected_replays'] == 12 and verification['test_predictions'] == 0 and
         all(verification[key] is True for key in ('training_complete','export_whole_file_byte_equal',
             'completed_paths_no_fitting_or_inference','campaign_bytes_and_mtimes_unchanged',
             'replay_bytes_and_mtimes_unchanged','within_campaign_deadline')),
         'Operational completion verification differs')
    need(sha(task/'operational_pipeline.py') == verification['operational_source_sha256'],
         'Completed operational helper differs')
    fits = read(campaign/'all_fits.json')
    need(len(fits) == 13 and sha(campaign/'all_fits.json') == status['all_fits_sha'],'Completed all-fits bytes differ')
    identities = [fit_name(row) for row in fits]
    need(len(set(identities)) == 13,'Duplicate fit identity')
    selection = read(campaign/'rate_selection.json');rate = selection['rate']
    need(rate in (.001,.003) and selection['plan_sha'] == PLAN_SHA,'Common rate binding differs')
    primary = [row for row in fits if row['rate'] == rate]
    need(len(primary) == 12 and {(row['control'],row['seed']) for row in primary} ==
         {(control,seed) for control in CONTROLS for seed in SEEDS},'Selected twelve-fit grid differs')
    need(audit['export_whole_file_byte_equal'] is True and
         verification['export_file_sha256'] == audit['export_file_sha256'],'Audited export inventories disagree')
    exports = inventory(task/'export_a_v1',30)
    need(set(exports) == EXPORT_NAMES and len(exports) == 17 and exports == audit['export_file_sha256'],
         'The seventeen export bytes differ from the independent audit')
    need(inventory(task/'export_b_v1',30) == exports,'The second export no longer matches')
    declared = read(task/'export_a_v1/artifact_sha256.json')
    need(declared == {name:value for name,value in exports.items() if name != 'artifact_sha256.json'},
         'Original export manifest differs')
    replay_files = inventory(replay,40)
    need(replay_files == audit['replay_artifact_sha256'],'Replay bytes differ from completed independent audit')
    need(not (replay/'failure.json').exists(),'Failed replay evidence must not be staged')
    complete = read(replay/'complete.json')
    need(complete['complete'] is True and complete['plan_sha'] == PLAN_SHA and complete['test_predictions'] == 0 and
         len(complete['records']) == 12 and complete['all_fits_sha'] == status['all_fits_sha'],
         'Selected replay completion differs')
    expected_records = {fit_name(row)+'.json' for row in primary}
    need({path.name for path in (replay/'records').iterdir()} == expected_records and
         {path.name for path in (replay/'intents').iterdir()} == expected_records,'Replay records/intents grid differs')
    # Every numeric campaign artifact staged below must have been covered by
    # the independent report. The checkpoint entries are deliberately not opened
    # or copied here: their recorded SHA values remain in the unchanged reports.
    campaign_bound = audit['campaign_artifact_sha256']
    payload = []
    def add(source,destination,expected=None):
        source = Path(source)
        need(source.is_file() and not source.is_symlink() and source.resolve().is_relative_to(task),
             'Allowlisted source is missing/linked/outside task: '+str(source))
        relative = Path(destination)
        need(not relative.is_absolute() and '..' not in relative.parts and relative.parts[0] not in ('data','figures'),
             'Unsafe or ignored publication destination')
        need(source.suffix not in ('.pt','.pth','.npz','.zip','.tif','.tiff') and
             (source.suffix != '.png' or relative == Path('exports/learning_curves.png')),
             'Weights, image tiles or raw numeric archives are excluded')
        actual = sha(source)
        if expected is not None:need(actual == expected,'Audited allowlist bytes differ: '+str(source))
        need(source.stat().st_size <= 32*1024*1024,'Allowlisted file exceeds packaging size bound')
        payload.append(dict(source=str(source),destination=str(relative),sha256=actual,
                            source_sha256=actual,destination_sha256=actual,bytes=source.stat().st_size))
    for name in sorted(EXPORT_NAMES):add(task/'export_a_v1'/name,'exports/'+name,exports[name])
    for name in ('plan.json','status.json','all_fits.json','rate_selection.json'):
        key='campaign_v1/'+name
        need(key in campaign_bound,'Campaign root artifact was not independently audited: '+key)
        add(campaign/name,'raw/'+key,campaign_bound[key])
    for row in fits:
        folder=fit_name(row)
        need(row['status'] == 'complete' and row['plan_sha'] == PLAN_SHA and row['source'] == plan['source'],
             'Completed fit binding differs')
        for name in ('result.json','status.json','history.json','training_history.json'):
            key='campaign_v1/'+folder+'/'+name
            need(key in campaign_bound,'Fit JSON was not independently audited: '+key)
            add(campaign/folder/name,'raw/'+key,campaign_bound[key])
    replay_allow = ('plan.json','complete.json','checkpoint_audit.json','artifact_sha256.json')
    for name in replay_allow:add(replay/name,'raw/replay_v1/'+name,replay_files[name])
    for folder in ('intents','records'):
        for name in sorted(expected_records):
            key=folder+'/'+name
            add(replay/key,'raw/replay_v1/'+key,replay_files[key])
    # Successful final evidence only. Failed/preflight attempts, private task
    # coordination and the full dataset/archive are intentionally outside scope.
    operations = {
        'pipeline_status.json':'pipeline_status.json',
        'pipeline_verification.json':'pipeline_verification.json',
        'operational_pipeline.py':'operational_pipeline.py.txt',
        'operational_pipeline.log':'operational_pipeline.log',
        'validation_replay.log':'validation_replay.log',
        'export_a.log':'export_a.log',
        'export_b.log':'export_b.log',
        'profile_v2.json':'profile.json',
        'profile_v2.log':'profile.log',
        'campaign_v1.log':'campaign.log',
    }
    need(sha(task/'profile_v2.json') == plan['profile_sha'],'Successful profile file differs from the frozen plan')
    for source,destination in operations.items():
        add(task/source,'operations/completed_v1/'+destination)
    for source,destination in {
        'numeric_audit/completed_v1/audit.json':'audit.json',
        'numeric_audit/completed_v1/REPORT.md':'REPORT.md',
        'numeric_audit/independent_numeric_audit.py':'independent_numeric_audit.py.txt',
        'numeric_audit/wait_for_pipeline.py':'wait_for_pipeline.py.txt',
        'numeric_audit/watcher_status.json':'watcher_status.json',
        'numeric_audit/actual_audit.log':'actual_audit.log',
        'numeric_audit/watcher.log':'watcher.log',
    }.items():add(task/source,'numeric_audit/completed_v1/'+destination)
    add(SELF,'operations/completed_v1/stage_completed_payload.py.txt')
    destinations=[row['destination'] for row in payload]
    need(len(destinations) == len(set(destinations)),'Duplicate publication destination')
    need(len(payload) <= 125 and sum(row['bytes'] for row in payload) <= 128*1024*1024,
         'Allowlisted payload exceeds the bounded publication scope')
    return payload,pipeline,status,watcher,audit


def stage(task,output):
    task,output=Path(task).resolve(),Path(output).resolve()
    allowed=task/'publication_preparation'
    need(task == TASK and output != allowed and output.is_relative_to(allowed),
         'Output must be a new child of the assigned task publication_preparation directory')
    need(not output.exists(),'Existing staging directory is preserved; choose a new output name')
    payload,pipeline,status,watcher,audit=make_allowlist(task)
    # No mutation occurs until every allowlisted source has passed admission and
    # byte validation. Exclusive creation makes an interrupted attempt sticky.
    output.mkdir()
    for row in payload:
        source,target=Path(row['source']),output/row['destination']
        need(sha(source) == row['sha256'],'Source changed before copying: '+str(source))
        target.parent.mkdir(parents=True,exist_ok=True)
        with source.open('rb') as original,target.open('xb') as copied:
            for block in iter(lambda:original.read(1024*1024),b''):copied.write(block)
        need(sha(source) == sha(target) == row['sha256'] and target.stat().st_size == row['bytes'],
             'Copied bytes differ: '+str(source))
    # Recheck admission and all staged inputs after copying, without scientific
    # imports or recomputing any model-selection or numerical outcome.
    gate(task)
    need(all(sha(row['source']) == row['sha256'] and sha(output/row['destination']) == row['sha256'] for row in payload),
         'Input/output bytes moved during staging')
    manifest=dict(format='landcover-native-publication-staging-v1',complete=True,plan_sha256=PLAN_SHA,
        packaging_source_sha256=sha(SELF),pipeline_verification_sha256=pipeline['verification_sha'],
        independent_audit_sha256=watcher['audit_sha'],independent_report_sha256=watcher['report_sha'],
        exported_files=17,fit_records=13,selected_validation_replays=12,test_predictions=0,
        all_payload_bytes_preserved=True,files=payload,
        exclusions=['checkpoint weights and .pt files','raw image/mask tiles and dataset tree','download archive',
                    'raw prediction arrays','lock files','failed/preflight attempt outputs','private coordination',
                    'new scientific conclusions, review decisions and publication actions'])
    with (output/'publication_manifest.json').open('x') as stream:
        stream.write(json.dumps(manifest,sort_keys=True,indent=2,allow_nan=False)+'\n')
    print(json.dumps(dict(status='staged',directory=str(output),files=len(payload),
        manifest_sha256=sha(output/'publication_manifest.json'),published=False),sort_keys=True))
    return output


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--task-root',type=Path,default=TASK)
    parser.add_argument('--output',type=Path,required=True)
    args=parser.parse_args()
    try:stage(args.task_root,args.output)
    except (ValueError,KeyError,FileNotFoundError) as error:
        parser.exit(3,str(error)+'\n')


if __name__ == '__main__':main()
