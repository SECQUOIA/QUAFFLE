def check(condition, message):
    checks.append(message)
    if not condition:

def load(path):
    def reject(value):
        raise ValueError('Nonfinite JSON constant: ' + value)
    return json.loads(path.read_text(), parse_constant=reject)


def sha(path):
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            digest.update(block)
    return digest.hexdigest()


def finite(value):
    if isinstance(value, float):
        return math.isfinite(value)
    if isinstance(value, dict):
        return all(finite(item) for item in value.values())
    if isinstance(value, list):
        return all(finite(item) for item in value)
    return True


def stats(values):
    return {'mean': statistics.mean(values), 'std': statistics.stdev(values), 'n': len(values)}


def close(actual, expected):
    return math.isclose(actual, expected, rel_tol=0, abs_tol=1e-12)


def metrics_check(metrics, n_images, label, optical_count=None):
    totals = {key: 0 for key in ('tp', 'fp', 'fn', 'tn')}
    if optical_count is None:
        check(n_images == 205, label + ': explicit validation sensor cohort required')
        optical_count = 71
    for sensor, n in (('optical', optical_count), ('sar', n_images-optical_count)):
        prefix = sensor + '_subset_'
        counts = {key: metrics[prefix + key] for key in totals}
        check(all(type(value) is int and value >= 0 for value in counts.values()), label + '/' + sensor + ': integer counts')
        check(metrics[prefix + 'n_samples'] == n and sum(counts.values()) == n * 128 * 128,
              label + '/' + sensor + ': cohort/pixel totals')
        tp, fp, fn, tn = (counts[key] for key in ('tp', 'fp', 'fn', 'tn'))
        denominator = tp + fp + fn
        calculated = {'iou': tp / denominator if denominator else 0.,
                      'dice': 2 * tp / (2 * tp + fp + fn) if 2 * tp + fp + fn else 0.,
                      'accuracy': (tp + tn) / sum(counts.values())}
        for metric, value in calculated.items():
            check(close(metrics[prefix + metric], value), label + '/' + sensor + ': count-derived ' + metric)
        check(0 <= metrics[prefix + 'auc'] <= 1, label + '/' + sensor + ': AUC range only')
        for key, value in counts.items():
            totals[key] += value
    tp, fp, fn, tn = (totals[key] for key in ('tp', 'fp', 'fn', 'tn'))
    denom = tp + fp + fn
    pooled = {'iou': tp / denom if denom else 0.,
              'dice': 2 * tp / (2 * tp + fp + fn) if 2 * tp + fp + fn else 0.,
              'accuracy': (tp + tn) / sum(totals.values())}
    check(sum(totals.values()) == n_images * 128 * 128, label + ': pooled pixels')
    for metric, value in pooled.items():
        check(close(metrics[metric], value), label + ': recombined pooled ' + metric)
    check(0 <= metrics['auc'] <= 1, label + ': pooled AUC range only')

def double_conv(c_in, c_out):
    return 9*c_in + c_in*c_out + 2*c_out + 9*c_out + c_out*c_out + 2*c_out


def up(c_in, c_out):
    return 4*c_in*c_out + c_out + double_conv(2*c_out, c_out)

    'stem_and_down': double_conv(3, 8) + double_conv(8, 16) + double_conv(16, 32) + double_conv(32, 64),

    'dense_reconstruction': 32*(128*8*8) + 128*8*8,
    'decoder': sum(up(a, b) for a, b in ((128, 64), (64, 32), (32, 16), (16, 8))),
